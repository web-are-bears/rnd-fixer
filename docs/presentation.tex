\begin{frame}{\insertsection: Constraints}
\begin{enumerate}
\item{Each student has a total of 100 tokens to be distributed among N projects.}
\item{A student must either allocate 0 tokens to a project, or a number between 5 and 25}
\item{Each professor has a total of 100 tokens PER PROJECT}
\item{Similarly, each project must either allocate 0 tokens to a project, or a number between 5 and 25}
\item{ALL 100 tokens must be allocated}
\end{enumerate}

S = {S1, S2, S3, ...} -> set of students
Pr = {Pr1, Pr2, ...}  -> set of professors
P = {P1, P2, P3, ...} -> set of projects

MATCHFIX(S, Pr, P) {
    while S != PHI:
        select some Si from S
        select most preferred Pi from P which was never picked for S
        if Pi has no slots left AND Si is more preferred than some Sj in Pi:
            pick the least preferred student Sk for Pi
            S = S union Sk
            add Si to Pi
            remove Sk from Pi
        else
            S = S - Si
            add Si to Pi
}
