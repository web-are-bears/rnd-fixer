\begin{frame}{\insertsection: Constraints}
\begin{enumerate}
\item{Each student has a total of 100 tokens to be distributed among N projects.}
\item{A student must either allocate 0 tokens to a project, or a number between 5 and 25}
\item{Each professor has a total of 100 tokens PER PROJECT}
\item{Similarly, each project must either allocate 0 tokens to a project, or a number between 5 and 25}
\item{ALL 100 tokens must be allocated}
\end{enumerate}
\end{frame}

\begin{frame}[fragile]
\frametitle{\insertsection: Algorithm}
\ttfamily
\begin{block}{Pseudocode}
\begin{verbatim} %looks horrible please fix
S = {S1, S2, S3, ...} -> set of students
Pr = {Pr1, Pr2, ...}  -> set of professors
P = {P1, P2, P3, ...} -> set of projects

MATCHFIX(S, Pr, P) {
    while S != PHI:
        select some Si from S
        select most preferred Pi from P which was never picked for S
        if Pi has no slots left AND Si is more preferred than some Sj in Pi:
            pick the least preferred student Sk for Pi
            S = S union Sk
            add Si to Pi
            remove Sk from Pi
        else
            S = S - Si
            add Si to Pi
}
\end{verbatim}
\end{block}

\begin{frame}{gRPC-Gateway}
  \small
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.47\textwidth}
    \textbf{Without it}\\[-2pt]
    {\footnotesize\color{black!60} hand-written gateway}
    \begin{itemize}\setlength{\itemsep}{6pt}
      \onslide<1-> \item A route + handler for \emph{every} RPC
      \onslide<2-> \item Each handler repeats: parse request, call gRPC, build JSON, map errors
      \onslide<3-> \item if proto changes edit gateway by hand
      \onslide<4-> \item Routes drift out of sync with services
    \end{itemize}
  \end{column}
  \begin{column}{0.47\textwidth}
    \textbf{With it}\\[-2pt]
    {\footnotesize\color{black!60} generated from \texttt{.proto}}
    \begin{itemize}\setlength{\itemsep}{6pt}
      \onslide<1-> \item Route declared in the \texttt{.proto}, next to the RPC
      \onslide<2-> \item \texttt{protoc} generates the handler (\texttt{*.pb.gw.go})
      \onslide<3-> \item Gateway just registers it + adds our middleware
      \onslide<4-> \item if proto changes regenerate, always in sync
    \end{itemize}
  \end{column}
\end{columns}
 
\vspace{18pt}
\begin{center}
  \color{accent}\textbf{New endpoint = edit \texttt{.proto} + regenerate}
\end{center}
\end{frame}
